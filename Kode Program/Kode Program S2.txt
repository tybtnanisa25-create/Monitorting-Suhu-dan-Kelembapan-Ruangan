#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>
#include <math.h>

// ======================================
// KONFIGURASI S2
// ======================================
#define DHTPIN 15
#define DHTTYPE DHT22

const char* ssid = "Wokwi-GUEST";
const char* password = "";
const char* server = "http://httpbin.org/post";

const int JUMLAH_PESAN = 100;
const int PAYLOAD_SIZE = 500;

const unsigned long INTERVAL = 2000;
const unsigned long TIMEOUT = 5000;

DHT dht(DHTPIN, DHTTYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Penyimpanan hasil
unsigned long rttData[JUMLAH_PESAN];
int rssiData[JUMLAH_PESAN];

int jumlahRTT = 0;
int jumlahRSSI = 0;

int berhasil = 0;
int gagal = 0;
int httpDicoba = 0;

bool selesai = false;

// ======================================
// MENAMPILKAN DATA CSV
// ======================================
void cetakCSV(
  int seq,
  bool adaKirim,
  unsigned long tKirim,
  bool sukses,
  unsigned long tTerima,
  int rssi
) {
  Serial.print("S2,");
  Serial.print(seq);
  Serial.print(",");
  Serial.print(PAYLOAD_SIZE);
  Serial.print(",");

  if (adaKirim) {
    Serial.print(tKirim);
  }

  Serial.print(",");

  if (sukses) {
    Serial.print(tTerima);
    Serial.print(",");
    Serial.print(tTerima - tKirim);
  } else {
    // Waktu terima dan RTT kosong
    Serial.print(",");
  }

  Serial.print(",");
  Serial.print(sukses ? "BERHASIL" : "GAGAL");
  Serial.print(",");
  Serial.println(rssi);
}

// ======================================
// TAMPILAN LCD
// ======================================
void tampilLCD(float suhu, float hum, int seq) {
  lcd.setCursor(0, 0);
  lcd.print("T:");
  lcd.print(suhu, 1);
  lcd.print("C H:");
  lcd.print(hum, 0);
  lcd.print("%    ");

  lcd.setCursor(0, 1);
  lcd.print("S2 ");
  lcd.print(seq);
  lcd.print("/100       ");
}

// ======================================
// STATISTIK RTT DAN RSSI
// ======================================
void hitungStatistik() {

  Serial.println();
  Serial.println("========== HASIL S2 ==========");

  Serial.print("Total pesan      : ");
  Serial.println(JUMLAH_PESAN);

  Serial.print("HTTP dicoba      : ");
  Serial.println(httpDicoba);

  Serial.print("Berhasil         : ");
  Serial.println(berhasil);

  Serial.print("Gagal            : ");
  Serial.println(gagal);

  Serial.print("PDR              : ");
  Serial.print(
    100.0 * berhasil / JUMLAH_PESAN, 2
  );
  Serial.println(" %");

  // ====================================
  // STATISTIK RTT
  // ====================================
  if (jumlahRTT > 0) {

    unsigned long urut[JUMLAH_PESAN];
    double totalRTT = 0;

    for (int i = 0; i < jumlahRTT; i++) {
      urut[i] = rttData[i];
      totalRTT += rttData[i];
    }

    // Urutkan data
    for (int i = 1; i < jumlahRTT; i++) {
      unsigned long nilai = urut[i];
      int j = i - 1;

      while (j >= 0 && urut[j] > nilai) {
        urut[j + 1] = urut[j];
        j--;
      }

      urut[j + 1] = nilai;
    }

    // Mean
    double mean = totalRTT / jumlahRTT;

    // Median
    double median;

    if (jumlahRTT % 2 == 0) {
      median =
        ((double)urut[jumlahRTT / 2 - 1] +
         urut[jumlahRTT / 2]) / 2.0;
    } else {
      median = urut[jumlahRTT / 2];
    }

    // Standar deviasi populasi
    double jumlahKuadrat = 0;

    for (int i = 0; i < jumlahRTT; i++) {
      double selisih =
        (double)rttData[i] - mean;

      jumlahKuadrat += selisih * selisih;
    }

    double sd =
      sqrt(jumlahKuadrat / jumlahRTT);

    // P95 metode nearest-rank
    int indeksP95 =
      (int)ceil(0.95 * jumlahRTT) - 1;

    Serial.print("RTT Mean         : ");
    Serial.print(mean, 2);
    Serial.println(" ms");

    Serial.print("RTT Median       : ");
    Serial.print(median, 2);
    Serial.println(" ms");

    Serial.print("RTT Std Dev      : ");
    Serial.print(sd, 2);
    Serial.println(" ms");

    Serial.print("RTT P95          : ");
    Serial.print(urut[indeksP95]);
    Serial.println(" ms");

    Serial.print("RTT Minimum      : ");
    Serial.print(urut[0]);
    Serial.println(" ms");

    Serial.print("RTT Maksimum     : ");
    Serial.print(urut[jumlahRTT - 1]);
    Serial.println(" ms");

  } else {
    Serial.println("Tidak ada RTT valid.");
  }

  // ====================================
  // STATISTIK RSSI
  // ====================================
  Serial.println("------------------------------");
  Serial.println("STATISTIK RSSI");

  if (jumlahRSSI > 0) {

    int minRSSI = rssiData[0];
    int maxRSSI = rssiData[0];
    double totalRSSI = 0;

    for (int i = 0; i < jumlahRSSI; i++) {

      int nilai = rssiData[i];
      totalRSSI += nilai;

      if (nilai < minRSSI) {
        minRSSI = nilai;
      }

      if (nilai > maxRSSI) {
        maxRSSI = nilai;
      }
    }

    double meanRSSI =
      totalRSSI / jumlahRSSI;

    Serial.print("Jumlah pengukuran: ");
    Serial.println(jumlahRSSI);

    Serial.print("RSSI Mean        : ");
    Serial.print(meanRSSI, 2);
    Serial.println(" dBm");

    Serial.print("RSSI Minimum     : ");
    Serial.print(minRSSI);
    Serial.println(" dBm");

    Serial.print("RSSI Maximum     : ");
    Serial.print(maxRSSI);
    Serial.println(" dBm");

  } else {
    Serial.println("Tidak ada data RSSI.");
  }

  Serial.println("==============================");
}

// ======================================
// SETUP
// ======================================
void setup() {

  Serial.begin(115200);

  Wire.begin(21, 22);
  lcd.init();
  lcd.backlight();

  dht.begin();

  lcd.setCursor(0, 0);
  lcd.print("IoT Experiment");
  lcd.setCursor(0, 1);
  lcd.print("S2 HTTP 500 B");

  delay(2000);

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password, 6);

  lcd.clear();
  lcd.print("Connecting WiFi");

  unsigned long mulai = millis();

  while (
    WiFi.status() != WL_CONNECTED &&
    millis() - mulai < 20000
  ) {
    delay(250);
  }

  if (WiFi.status() != WL_CONNECTED) {

    Serial.println("# WiFi gagal!");

    lcd.clear();
    lcd.print("WiFi GAGAL");

    while (true) {
      delay(1000);
    }
  }

  lcd.clear();
  lcd.print("WiFi Connected");
  delay(1000);

  Serial.println(
    "skenario,seq,payload_byte,t_kirim_ms,"
    "t_terima_ms,rtt_ms,status,rssi_dbm"
  );
}

// ======================================
// LOOP EKSPERIMEN S2
// ======================================
void loop() {

  if (selesai) return;

  for (int seq = 1; seq <= JUMLAH_PESAN; seq++) {

    // Baca RSSI setiap percobaan
    int rssi = WiFi.RSSI();

    rssiData[jumlahRSSI] = rssi;
    jumlahRSSI++;

    // Cek WiFi
    if (WiFi.status() != WL_CONNECTED) {

      gagal++;
      cetakCSV(seq, false, 0, false, 0, rssi);

      if (seq < JUMLAH_PESAN)
        delay(INTERVAL);

      continue;
    }

    // Baca sensor
    float suhu = dht.readTemperature();
    float hum = dht.readHumidity();

    if (isnan(suhu) || isnan(hum)) {

      gagal++;
      cetakCSV(seq, false, 0, false, 0, rssi);

      if (seq < JUMLAH_PESAN)
        delay(INTERVAL);

      continue;
    }

    tampilLCD(suhu, hum, seq);

    // Buat token identitas
    char token[32];

    snprintf(
      token,
      sizeof(token),
      "S%04dT%03dH%03d",
      seq,
      (int)round(suhu),
      (int)round(hum)
    );

    String payload = String(token);

    // Panjang payload tepat 500 byte
    while (payload.length() < PAYLOAD_SIZE) {
      payload += "X";
    }

    if (payload.length() != PAYLOAD_SIZE) {

      gagal++;
      cetakCSV(seq, false, 0, false, 0, rssi);

      if (seq < JUMLAH_PESAN)
        delay(INTERVAL);

      continue;
    }

    // HTTP POST
    HTTPClient http;

    http.begin(server);
    http.addHeader(
      "Content-Type", "text/plain"
    );
    http.setTimeout(TIMEOUT);

    unsigned long tKirim = millis();
    httpDicoba++;

    int kode = http.POST(
      (uint8_t*)payload.c_str(),
      payload.length()
    );

    String respons = "";

    if (kode > 0) {
      respons = http.getString();
    }

    unsigned long tTerima = millis();
    unsigned long rtt = tTerima - tKirim;

    bool sukses =
      kode >= 200 &&
      kode < 300 &&
      respons.indexOf(token) >= 0 &&
      rtt <= TIMEOUT;

    if (sukses) {
      berhasil++;
      rttData[jumlahRTT++] = rtt;
    } else {
      gagal++;
    }

    cetakCSV(
      seq, true, tKirim,
      sukses, tTerima, rssi
    );

    http.end();

    if (seq < JUMLAH_PESAN) {
      delay(INTERVAL);
    }
  }

  // Ringkasan setelah 100 pesan
  hitungStatistik();

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("S2 SELESAI");

  lcd.setCursor(0, 1);
  lcd.print("PDR:");
  lcd.print(
    100.0 * berhasil / JUMLAH_PESAN,
    1
  );
  lcd.print("%");

  selesai = true;
}
