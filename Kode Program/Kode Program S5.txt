#include <WiFi.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>
#include <math.h>

// Konfigurasi LCD I2C
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Konfigurasi DHT22
#define DHTPIN 15
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);

// Konfigurasi WiFi
const char* ssid = "Wokwi-GUEST";
const char* password = "";

// Konfigurasi MQTT
const char* broker = "test.mosquitto.org";
const int port = 1883;
const char* topic = "telkom/s5/monitoring";

// Konfigurasi eksperimen
#define JUMLAH_PESAN 100
#define INTERVAL 2000
#define TIMEOUT 5000
#define PAYLOAD_SIZE 20

WiFiClient client;

int berhasil = 0;
int gagal = 0;
int jumlahRTT = 0;

float rssiData[JUMLAH_PESAN];
float rttData[JUMLAH_PESAN];

uint16_t packetId = 1;

// Mengirim Remaining Length MQTT
void sendRemainingLength(uint32_t len) {
  do {
    uint8_t digit = len % 128;
    len /= 128;

    if (len > 0) {
      digit |= 0x80;
    }

    client.write(digit);
  } while (len > 0);
}

// Koneksi MQTT
bool connectMQTT() {
  if (!client.connect(broker, port)) {
    return false;
  }

  uint8_t header[] = {
    0x10, 0x10,
    0x00, 0x04, 'M', 'Q', 'T', 'T',
    0x04, 0x02, 0x00, 0x3C,
    0x00, 0x04, 'S', '5', 'E', 'S'
  };

  client.write(header, sizeof(header));

  unsigned long start = millis();

  while (millis() - start < 5000) {
    if (client.available() >= 4) {
      uint8_t response[4];
      client.read(response, 4);

      return response[0] == 0x20 &&
             response[3] == 0x00;
    }
  }

  return false;
}

// Mengirim MQTT PUBLISH QoS 1
void publishQoS1(String payload, uint16_t id) {
  client.write((uint8_t)0x32);

  uint32_t remaining =
    2 + strlen(topic) + 2 + payload.length();

  sendRemainingLength(remaining);

  uint8_t topicLength = strlen(topic);

  client.write((uint8_t)0);
  client.write(topicLength);

  client.write(
    (const uint8_t*)topic,
    topicLength
  );

  client.write((uint8_t)(id >> 8));
  client.write((uint8_t)(id & 0xFF));

  client.write(
    (const uint8_t*)payload.c_str(),
    payload.length()
  );
}

// Menunggu MQTT PUBACK
bool waitPUBACK(
  uint16_t id,
  unsigned long start,
  unsigned long &tTerima,
  float &rtt
) {
  unsigned long timeoutStart = millis();

  while (millis() - timeoutStart < TIMEOUT) {
    if (client.available() >= 4) {
      uint8_t packet[4];
      client.read(packet, 4);

      if (packet[0] == 0x40 &&
          packet[1] == 0x02) {

        uint16_t receivedId =
          (packet[2] << 8) | packet[3];

        if (receivedId == id) {
          tTerima = millis();
          rtt = tTerima - start;
          return true;
        }
      }
    }

    delay(1);
  }

  return false;
}

// Tampilan LCD
void tampilkanLCD(int seq, float rtt, bool status) {
  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("SEQ: ");
  lcd.print(seq);
  lcd.print("/100");

  lcd.setCursor(0, 1);

  if (status) {
    lcd.print("RTT:");
    lcd.print(rtt, 0);
    lcd.print("ms OK");
  } else {
    lcd.print("MQTT GAGAL");
  }
}

// Menampilkan rekapitulasi hasil
void tampilkanHasil() {
  float totalRSSI = 0;
  float minRSSI = rssiData[0];
  float maxRSSI = rssiData[0];

  for (int i = 0; i < JUMLAH_PESAN; i++) {
    totalRSSI += rssiData[i];

    if (rssiData[i] < minRSSI) {
      minRSSI = rssiData[i];
    }

    if (rssiData[i] > maxRSSI) {
      maxRSSI = rssiData[i];
    }
  }

  float meanRSSI = totalRSSI / JUMLAH_PESAN;

  Serial.println();
  Serial.println("========== HASIL S5 ==========");
  Serial.printf("Total pesan    : %d\n", JUMLAH_PESAN);
  Serial.printf("Berhasil       : %d\n", berhasil);
  Serial.printf("Gagal          : %d\n", gagal);

  Serial.printf(
    "PDR            : %.2f %%\n",
    berhasil * 100.0 / JUMLAH_PESAN
  );

  if (jumlahRTT > 0) {
    float totalRTT = 0;

    for (int i = 0; i < jumlahRTT; i++) {
      totalRTT += rttData[i];
    }

    float meanRTT = totalRTT / jumlahRTT;

    // Urutkan data RTT
    for (int i = 0; i < jumlahRTT - 1; i++) {
      for (int j = i + 1; j < jumlahRTT; j++) {
        if (rttData[i] > rttData[j]) {
          float temp = rttData[i];
          rttData[i] = rttData[j];
          rttData[j] = temp;
        }
      }
    }

    // Median
    float median;

    if (jumlahRTT % 2 == 0) {
      median = (
        rttData[jumlahRTT / 2 - 1] +
        rttData[jumlahRTT / 2]
      ) / 2.0;
    } else {
      median = rttData[jumlahRTT / 2];
    }

    // Standar deviasi populasi
    float variance = 0;

    for (int i = 0; i < jumlahRTT; i++) {
      variance += pow(rttData[i] - meanRTT, 2);
    }

    float stdDev = sqrt(variance / jumlahRTT);

    // P95
    int indexP95 = (int)ceil(0.95 * jumlahRTT) - 1;
    float p95 = rttData[indexP95];

    Serial.printf("RTT Mean       : %.2f ms\n", meanRTT);
    Serial.printf("RTT Median     : %.2f ms\n", median);
    Serial.printf("RTT Std Dev    : %.2f ms\n", stdDev);
    Serial.printf("RTT P95        : %.0f ms\n", p95);

  } else {
    Serial.println("RTT Mean       : Tidak tersedia");
    Serial.println("RTT Median     : Tidak tersedia");
    Serial.println("RTT Std Dev    : Tidak tersedia");
    Serial.println("RTT P95        : Tidak tersedia");
  }

  Serial.println("--------------------------------");
  Serial.println("STATISTIK RSSI");
  Serial.printf("Jumlah pengukuran : %d\n", JUMLAH_PESAN);
  Serial.printf("RSSI Mean         : %.2f dBm\n", meanRSSI);
  Serial.printf("RSSI Minimum      : %.0f dBm\n", minRSSI);
  Serial.printf("RSSI Maximum      : %.0f dBm\n", maxRSSI);
  Serial.println("===============================");

  // Tampilan akhir LCD
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("S5 SELESAI");

  lcd.setCursor(0, 1);
  lcd.print("PDR:");
  lcd.print(berhasil);
  lcd.print("%");
}

// Setup
void setup() {
  Serial.begin(115200);

  // Inisialisasi LCD
  Wire.begin(21, 22);
  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print("S5 MQTT QoS 1");

  lcd.setCursor(0, 1);
  lcd.print("Connecting WiFi");

  dht.begin();

  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("WiFi Connected");

  lcd.setCursor(0, 1);
  lcd.print("Connecting MQTT");

  if (!connectMQTT()) {
    Serial.println("MQTT connection failed!");

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("MQTT FAILED");
    return;
  }

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("S5 MQTT QoS 1");

  lcd.setCursor(0, 1);
  lcd.print("Payload: 20 B");

  Serial.println(
    "skenario,seq,payload_byte,t_kirim_ms,"
    "t_terima_ms,rtt_ms,status,rssi_dbm"
  );
}

// Loop eksperimen
void loop() {
  static bool selesai = false;

  if (selesai) {
    return;
  }

  for (int i = 0; i < JUMLAH_PESAN; i++) {
    delay(INTERVAL);

    float rssi = WiFi.RSSI();
    rssiData[i] = rssi;

    // Periksa koneksi MQTT
    if (!client.connected()) {
      if (!connectMQTT()) {
        gagal++;

        Serial.printf(
          "S5,%d,%d,,,,GAGAL,%.0f\n",
          i + 1,
          PAYLOAD_SIZE,
          rssi
        );

        tampilkanLCD(i + 1, 0, false);
        continue;
      }
    }

    // Membuat payload 20 byte
    String token = "S5-" + String(i + 1);
    String payload = token;

    while (payload.length() < PAYLOAD_SIZE) {
      payload += "X";
    }

    payload = payload.substring(0, PAYLOAD_SIZE);

    // Waktu pengiriman
    float rtt = 0;
    unsigned long tKirim = millis();
    unsigned long tTerima = 0;

    // Mengirim pesan
    publishQoS1(payload, packetId);

    // Menunggu PUBACK
    if (waitPUBACK(
          packetId,
          tKirim,
          tTerima,
          rtt
        )) {

      berhasil++;

      rttData[jumlahRTT] = rtt;
      jumlahRTT++;

      Serial.printf(
        "S5,%d,%d,%lu,%lu,%.0f,BERHASIL,%.0f\n",
        i + 1,
        PAYLOAD_SIZE,
        tKirim,
        tTerima,
        rtt,
        rssi
      );

      tampilkanLCD(i + 1, rtt, true);

    } else {
      gagal++;

      Serial.printf(
        "S5,%d,%d,%lu,,,GAGAL,%.0f\n",
        i + 1,
        PAYLOAD_SIZE,
        tKirim,
        rssi
      );

      tampilkanLCD(i + 1, 0, false);
    }

    // Packet ID berikutnya
    packetId++;

    if (packetId == 0) {
      packetId = 1;
    }
  }

  tampilkanHasil();
  selesai = true;
}
