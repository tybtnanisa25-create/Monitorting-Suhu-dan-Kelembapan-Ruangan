#include <WiFi.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>
#include <MQTT.h>
#include <math.h>
#include <esp_system.h>

// ======================================
// KONFIGURASI S4
// ======================================
#define DHTPIN 15
#define DHTTYPE DHT22

const char* ssid = "Wokwi-GUEST";
const char* password = "";

const char* broker = "test.mosquitto.org";
const int port = 1883;

const int JUMLAH_PESAN = 100;
const int PAYLOAD_SIZE = 500;
const int QOS = 0;

const unsigned long INTERVAL = 2000;
const unsigned long TIMEOUT = 5000;

DHT dht(DHTPIN, DHTTYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2);

WiFiClient net;
MQTTClient mqtt(2048);

String topik;

// ======================================
// DATA PENGUKURAN
// ======================================
unsigned long rttData[JUMLAH_PESAN];
int rssiData[JUMLAH_PESAN];

int jumlahRTT = 0;
int jumlahRSSI = 0;

int berhasil = 0;
int gagal = 0;
int publishDicoba = 0;

bool menunggu = false;
bool selesai = false;

int seqAktif = 0;
int rssiAktif = 0;

unsigned long tKirimAktif = 0;

// ======================================
// OUTPUT CSV
// ======================================
void cetakCSV(
  int seq,
  bool adaKirim,
  unsigned long tKirim,
  bool sukses,
  unsigned long tTerima,
  int rssi
) {
  Serial.print("S4,");
  Serial.print(seq);
  Serial.print(",");
  Serial.print(PAYLOAD_SIZE);
  Serial.print(",");

  if (adaKirim) {
    Serial.print(tKirim);
  }

  Serial.print(",");

  if (sukses) {
    Serial.print(tTerima);
    Serial.print(",");
    Serial.print(tTerima - tKirim);
  } else {
    // Kolom waktu terima dan RTT kosong
    Serial.print(",");
  }

  Serial.print(",");
  Serial.print(sukses ? "BERHASIL" : "GAGAL");
  Serial.print(",");
  Serial.println(rssi);
}

// ======================================
// TAMPILAN LCD
// ======================================
void tampilLCD(float suhu, float hum, int seq) {

  lcd.setCursor(0, 0);
  lcd.print("T:");
  lcd.print(suhu, 1);
  lcd.print("C H:");
  lcd.print(hum, 0);
  lcd.print("%    ");

  lcd.setCursor(0, 1);
  lcd.print("S4 ");
  lcd.print(seq);
  lcd.print("/100       ");
}

// ======================================
// CALLBACK PESAN MQTT
// ======================================
// QoS 0 tidak menggunakan PUBACK.
// RTT diukur sampai pesan diterima
// kembali melalui subscription.

void pesanDiterima(
  String &topic,
  String &message
) {
  if (!menunggu) return;
  if (topic != topik) return;

  char prefix[20];

  snprintf(
    prefix,
    sizeof(prefix),
    "S4:%03d:",
    seqAktif
  );

  if (!message.startsWith(prefix)) return;

  if (message.length() != PAYLOAD_SIZE) {
    return;
  }

  unsigned long tTerima = millis();
  unsigned long rtt = tTerima - tKirimAktif;

  if (rtt > TIMEOUT) return;

  berhasil++;

  rttData[jumlahRTT++] = rtt;

  cetakCSV(
    seqAktif,
    true,
    tKirimAktif,
    true,
    tTerima,
    rssiAktif
  );

  menunggu = false;
}

// ======================================
// KONEKSI MQTT
// ======================================
bool sambungMQTT() {

  if (mqtt.connected()) return true;

  String clientId =
    "ESP32-S4-" +
    String((uint32_t)ESP.getEfuseMac(), HEX) +
    "-" +
    String((uint32_t)esp_random(), HEX);

  if (!mqtt.connect(clientId.c_str())) {
    return false;
  }

  return mqtt.subscribe(topik.c_str(), QOS);
}

// ======================================
// MEMBUAT PAYLOAD 500 BYTE
// ======================================
String buatPayload(
  int seq,
  float suhu,
  float hum
) {
  char token[32];

  snprintf(
    token,
    sizeof(token),
    "S4:%03d:T%03dH%03d",
    seq,
    (int)round(suhu),
    (int)round(hum)
  );

  String payload = String(token);

  while (payload.length() < PAYLOAD_SIZE) {
    payload += "X";
  }

  if (payload.length() != PAYLOAD_SIZE) {
    return "";
  }

  return payload;
}

// ======================================
// STATISTIK RTT DAN RSSI
// ======================================
void hitungStatistik() {

  Serial.println();
  Serial.println("========== HASIL S4 ==========");

  Serial.print("Total pesan      : ");
  Serial.println(JUMLAH_PESAN);

  Serial.print("Publish dicoba   : ");
  Serial.println(publishDicoba);

  Serial.print("Berhasil         : ");
  Serial.println(berhasil);

  Serial.print("Gagal            : ");
  Serial.println(gagal);

  Serial.print("PDR              : ");
  Serial.print(
    100.0 * berhasil / JUMLAH_PESAN, 2
  );
  Serial.println(" %");

  // ====================================
  // STATISTIK RTT
  // ====================================
  if (jumlahRTT > 0) {

    unsigned long urut[JUMLAH_PESAN];
    double totalRTT = 0;

    for (int i = 0; i < jumlahRTT; i++) {
      urut[i] = rttData[i];
      totalRTT += rttData[i];
    }

    // Urutkan data RTT
    for (int i = 1; i < jumlahRTT; i++) {

      unsigned long nilai = urut[i];
      int j = i - 1;

      while (j >= 0 && urut[j] > nilai) {
        urut[j + 1] = urut[j];
        j--;
      }

      urut[j + 1] = nilai;
    }

    // Mean
    double mean = totalRTT / jumlahRTT;

    // Median
    double median;

    if (jumlahRTT % 2 == 0) {
      median =
        ((double)urut[jumlahRTT / 2 - 1] +
         urut[jumlahRTT / 2]) / 2.0;
    } else {
      median = urut[jumlahRTT / 2];
    }

    // Standar deviasi populasi
    double jumlahKuadrat = 0;

    for (int i = 0; i < jumlahRTT; i++) {
      double selisih =
        (double)rttData[i] - mean;

      jumlahKuadrat += selisih * selisih;
    }

    double sd =
      sqrt(jumlahKuadrat / jumlahRTT);

    // P95 metode nearest-rank
    int indeksP95 =
      (int)ceil(0.95 * jumlahRTT) - 1;

    Serial.print("RTT Mean         : ");
    Serial.print(mean, 2);
    Serial.println(" ms");

    Serial.print("RTT Median       : ");
    Serial.print(median, 2);
    Serial.println(" ms");

    Serial.print("RTT Std Dev      : ");
    Serial.print(sd, 2);
    Serial.println(" ms");

    Serial.print("RTT P95          : ");
    Serial.print(urut[indeksP95]);
    Serial.println(" ms");

    Serial.print("RTT Minimum      : ");
    Serial.print(urut[0]);
    Serial.println(" ms");

    Serial.print("RTT Maksimum     : ");
    Serial.print(urut[jumlahRTT - 1]);
    Serial.println(" ms");

  } else {
    Serial.println("Tidak ada RTT valid.");
  }

  // ====================================
  // STATISTIK RSSI
  // ====================================
  Serial.println("------------------------------");
  Serial.println("STATISTIK RSSI");

  if (jumlahRSSI > 0) {

    int minRSSI = rssiData[0];
    int maxRSSI = rssiData[0];

    double totalRSSI = 0;

    for (int i = 0; i < jumlahRSSI; i++) {

      int nilai = rssiData[i];

      totalRSSI += nilai;

      if (nilai < minRSSI) {
        minRSSI = nilai;
      }

      if (nilai > maxRSSI) {
        maxRSSI = nilai;
      }
    }

    double meanRSSI =
      totalRSSI / jumlahRSSI;

    Serial.print("Jumlah pengukuran: ");
    Serial.println(jumlahRSSI);

    Serial.print("RSSI Mean        : ");
    Serial.print(meanRSSI, 2);
    Serial.println(" dBm");

    Serial.print("RSSI Minimum     : ");
    Serial.print(minRSSI);
    Serial.println(" dBm");

    Serial.print("RSSI Maximum     : ");
    Serial.print(maxRSSI);
    Serial.println(" dBm");

  } else {
    Serial.println("Tidak ada data RSSI.");
  }

  Serial.println("==============================");
}

// ======================================
// SETUP
// ======================================
void setup() {

  Serial.begin(115200);

  Wire.begin(21, 22);
  lcd.init();
  lcd.backlight();

  dht.begin();

  lcd.setCursor(0, 0);
  lcd.print("IoT Experiment");

  lcd.setCursor(0, 1);
  lcd.print("S4 MQTT QoS 0");

  delay(2000);

  // Koneksi WiFi
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password, 6);

  lcd.clear();
  lcd.print("Connecting WiFi");

  unsigned long mulai = millis();

  while (
    WiFi.status() != WL_CONNECTED &&
    millis() - mulai < 20000
  ) {
    delay(250);
  }

  if (WiFi.status() != WL_CONNECTED) {

    Serial.println("# WiFi gagal!");

    lcd.clear();
    lcd.print("WiFi GAGAL");

    while (true) {
      delay(1000);
    }
  }

  // Mulai MQTT
  mqtt.begin(broker, port, net);
  mqtt.onMessage(pesanDiterima);

  // Topik unik agar tidak tercampur
  // dengan pengguna broker lain
  topik =
    "telkom/iot/s4/" +
    String((uint32_t)ESP.getEfuseMac(), HEX) +
    "/" +
    String((uint32_t)esp_random(), HEX);

  lcd.clear();
  lcd.print("WiFi Connected");

  delay(1000);

  Serial.println(
    "skenario,seq,payload_byte,t_kirim_ms,"
    "t_terima_ms,rtt_ms,status,rssi_dbm"
  );
}

// ======================================
// LOOP EKSPERIMEN S4
// ======================================
void loop() {

  if (selesai) return;

  for (int seq = 1; seq <= JUMLAH_PESAN; seq++) {

    // Baca RSSI setiap percobaan
    int rssi = WiFi.RSSI();

    rssiData[jumlahRSSI] = rssi;
    jumlahRSSI++;

    // Baca sensor
    float suhu = dht.readTemperature();
    float hum = dht.readHumidity();

    // Validasi koneksi dan sensor
    if (
      WiFi.status() != WL_CONNECTED ||
      isnan(suhu) ||
      isnan(hum)
    ) {

      gagal++;

      cetakCSV(
        seq, false, 0, false, 0, rssi
      );

      if (seq < JUMLAH_PESAN)
        delay(INTERVAL);

      continue;
    }

    tampilLCD(suhu, hum, seq);

    // Pastikan MQTT sudah tersambung
    if (!sambungMQTT()) {

      gagal++;

      cetakCSV(
        seq, false, 0, false, 0, rssi
      );

      if (seq < JUMLAH_PESAN)
        delay(INTERVAL);

      continue;
    }

    mqtt.loop();

    // Membuat payload 500 byte
    String payload = buatPayload(
      seq, suhu, hum
    );

    if (payload.length() != PAYLOAD_SIZE) {

      gagal++;

      cetakCSV(
        seq, false, 0, false, 0, rssi
      );

      if (seq < JUMLAH_PESAN)
        delay(INTERVAL);

      continue;
    }

    // Menandai pesan aktif
    seqAktif = seq;
    rssiAktif = rssi;

    tKirimAktif = millis();
    menunggu = true;

    publishDicoba++;

    // PUBLISH MQTT QoS 0
    bool dikirim = mqtt.publish(
      topik.c_str(),
      payload.c_str(),
      false,
      QOS
    );

    // Gagal memulai publish
    if (!dikirim && menunggu) {

      menunggu = false;
      gagal++;

      cetakCSV(
        seq, true, tKirimAktif,
        false, 0, rssi
      );
    }

    // Menunggu echo dari broker
    while (
      menunggu &&
      millis() - tKirimAktif < TIMEOUT
    ) {

      if (!mqtt.connected()) break;

      mqtt.loop();
      delay(5);
    }

    // Timeout / koneksi terputus
    if (menunggu) {

      menunggu = false;
      gagal++;

      cetakCSV(
        seq, true, tKirimAktif,
        false, 0, rssi
      );
    }

    // Jeda 2 detik antartransaksi
    if (seq < JUMLAH_PESAN) {
      delay(INTERVAL);
    }
  }

  hitungStatistik();

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("S4 SELESAI");

  lcd.setCursor(0, 1);
  lcd.print("PDR:");
  lcd.print(
    100.0 * berhasil / JUMLAH_PESAN,
    1
  );
  lcd.print("%");

  selesai = true;
}
